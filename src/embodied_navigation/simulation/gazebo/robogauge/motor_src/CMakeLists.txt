cmake_minimum_required(VERSION 3.16)
project(robogauge_actuator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(robogauge_actuator SHARED go2hv_actuator.cpp)
target_link_libraries(robogauge_actuator PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS robogauge_actuator LIBRARY DESTINATION lib)
